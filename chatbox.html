// TUILayout.java — 主布局，用 Lanterna 实现左右分栏
public class TUILayout {

  private final Screen        screen;
  private final ContactPanel  contactPanel;
  private final ChatPanel     chatPanel;
  private final InputBar      inputBar;
  private       FocusTarget   focus = FocusTarget.CONTACT;

  public TUILayout(IMService service) {
    Terminal terminal = new DefaultTerminalFactory().createTerminal();
    screen = new TerminalScreen(terminal);
    screen.startScreen();

    int W = screen.getTerminalSize().getColumns();
    int H = screen.getTerminalSize().getRows();
    int leftW = 28;  // 左侧联系人面板宽度

    contactPanel = new ContactPanel(screen,
        new TerminalRectangle(0, 0, leftW, H-3), service);

    chatPanel = new ChatPanel(screen,
        new TerminalRectangle(leftW+1, 0, W-leftW-1, H-3), service);

    inputBar = new InputBar(screen,
        new TerminalRectangle(leftW+1, H-3, W-leftW-1, 3));
  }

  public void run() throws IOException {
    render();
    while (true) {
      KeyStroke key = screen.readInput();
      if (isQuit(key)) break;
      handleKey(key);
      render();
    }
    screen.stopScreen();
  }

  private void handleKey(KeyStroke key) {
    // Tab 切换焦点
    if (key.getKeyType() == KeyType.Tab) {
      focus = focus.next(); return;
    }
    switch (focus) {
      case CONTACT -> contactPanel.handleKey(key);
      case CHAT    -> chatPanel.handleKey(key);
      case INPUT   -> inputBar.handleKey(key);
    }
  }

  private void render() throws IOException {
    screen.clear();
    drawDivider();       // 中间竖线
    contactPanel.render(focus == FocusTarget.CONTACT);
    chatPanel.render(focus == FocusTarget.CHAT);
    inputBar.render(focus == FocusTarget.INPUT);
    screen.refresh();
  }
}
// ContactPanel.java — 左侧 Tab 联系人面板
public class ContactPanel {

  private final Screen           screen;
  private final TerminalRectangle rect;
  private final IMService         service;
  private       TabType           tab      = TabType.FRIENDS;
  private       int              selected = 0;
  private       int              scrollY  = 0;

  public void render(boolean focused) {
    TextGraphics g = screen.newTextGraphics();
    drawBorder(g, focused);

    // ① 顶部 Tab 栏
    drawTabs(g);

    // ② 联系人列表（虚拟滚动）
    List<Contact> list = getList();
    int visRows = rect.height() - 3;
    for (int i = scrollY;
         i < list.size() && (i-scrollY) < visRows; i++) {
      boolean sel = (i == selected);
      drawContactRow(g, list.get(i),
          rect.top() + 3 + (i-scrollY), sel);
    }
  }

  private void drawContactRow(TextGraphics g,
      Contact c, int y, boolean selected) {
    if (selected)
      g.setBackgroundColor(TextColor.rgb(40,50,70));
    // 状态点
    g.setForegroundColor(statusColor(c.status()));
    g.putString(rect.left()+1, y, "●");
    // 昵称
    g.setForegroundColor(nameColor(selected));
    g.putString(rect.left()+3, y,
        truncate(c.name(), 14));
    // 未读角标
    if (c.unread() > 0) {
      g.setForegroundColor(TextColor.ANSI.GREEN);
      g.putString(rect.right()-4, y,
          "["+c.unread()+"]");
    }
  }

  public void handleKey(KeyStroke key) {
    List<Contact> list = getList();
    switch (key.getKeyType()) {
      case ArrowUp   ->  { if(selected>0) selected--; adjustScroll(); }
      case ArrowDown ->  { if(selected<list.size()-1) selected++; adjustScroll(); }
      case Enter     -> service.openChat(list.get(selected));
      case Character -> {
        if(key.getCharacter() == '1') tab = TabType.FRIENDS;
        if(key.getCharacter() == '2') tab = TabType.GROUPS;
        if(key.getCharacter() == '3') tab = TabType.RECENT;
      }
    }
  }
}
// ChatPanel.java — 右侧消息区（带虚拟滚动）
public class ChatPanel {

  private final Screen           screen;
  private final TerminalRectangle rect;
  private final IMService         service;
  private       int               scrollOffset = 0;
  private       boolean           atBottom     = true;

  public void render(boolean focused) {
    TextGraphics g = screen.newTextGraphics();
    drawHeader(g);                   // 联系人标题栏

    List<Message> msgs = service.currentMessages();
    List<RenderLine> lines = wrap(msgs); // 自动折行
    int visible = rect.height() - 2;

    // 新消息到来时自动滚到底部
    if (atBottom)
      scrollOffset = Math.max(0, lines.size() - visible);

    for (int i = scrollOffset;
         i < lines.size() && (i-scrollOffset) < visible; i++) {
      drawLine(g, lines.get(i),
               rect.top()+2+(i-scrollOffset));
    }
    drawScrollBar(g, lines.size(), visible);
  }

  private void drawLine(TextGraphics g,
      RenderLine line, int y) {
    if (line.isSelf()) {
      // 自己的消息右对齐，绿色
      g.setForegroundColor(TextColor.ANSI.GREEN);
      int x = rect.right() - line.text().length() - 2;
      g.putString(x, y, line.text());
    } else {
      // 对方消息左对齐，白色
      g.setForegroundColor(TextColor.ANSI.WHITE);
      g.putString(rect.left()+1, y, line.text());
    }
  }

  public void handleKey(KeyStroke key) {
    switch (key.getKeyType()) {
      case ArrowUp   -> { scrollOffset--; atBottom = false; }
      case ArrowDown -> { scrollOffset++; checkAtBottom(); }
      case End       -> atBottom = true;
    }
  }
}
// InputBar.java — 底部输入栏（JLine 内嵌）
public class InputBar {

  private final Screen           screen;
  private final TerminalRectangle rect;
  private final StringBuilder    buffer    = new StringBuilder();
  private       int              cursor    = 0;
  private       Runnable         onSend;

  public void render(boolean focused) {
    TextGraphics g = screen.newTextGraphics();
    // 边框：激活时绿色高亮
    g.setForegroundColor(focused
        ? TextColor.ANSI.GREEN
        : TextColor.ANSI.WHITE);
    drawBox(g);

    // 提示符 + 缓冲内容
    g.setForegroundColor(TextColor.ANSI.CYAN);
    g.putString(rect.left()+1,
                rect.top()+1, "▶ ");

    g.setForegroundColor(TextColor.ANSI.WHITE);
    g.putString(rect.left()+4,
                rect.top()+1,
                buffer.toString());

    // 光标闪烁（偶数秒显示）
    if (focused && (System.currentTimeMillis() / 500) % 2 == 0) {
      g.setForegroundColor(TextColor.ANSI.GREEN);
      g.putString(rect.left()+4+cursor,
                  rect.top()+1, "█");
    }

    // 右下角快捷键提示
    g.setForegroundColor(TextColor.rgb(80,90,110));
    g.putString(rect.right()-20,
                rect.top()+2,
                "Enter:发送 Esc:清空");
  }

  public void handleKey(KeyStroke key) {
    switch (key.getKeyType()) {
      case Enter     -> { onSend.run(); buffer.setLength(0); cursor=0; }
      case Backspace  -> { if(cursor>0) { buffer.deleteCharAt(--cursor); } }
      case ArrowLeft  -> { if(cursor>0) cursor--; }
      case ArrowRight -> { if(cursor<buffer.length()) cursor++; }
      case Escape     -> { buffer.setLength(0); cursor=0; }
      case Character  -> {
        buffer.insert(cursor++, key.getCharacter());
      }
    }
  }
}
// IMRenderer.java — 统一渲染工具：边框、标题、分隔线
public class IMRenderer {

  /** 绘制圆角边框，焦点时绿色高亮 */
  public static void drawBox(TextGraphics g,
      TerminalRectangle r, boolean focused,
      String title) {

    g.setForegroundColor(focused
      ? TextColor.ANSI.GREEN
      : TextColor.rgb(60,70,90));

    // 四角
    g.putString(r.left(),    r.top(),    "╭");
    g.putString(r.right()-1, r.top(),    "╮");
    g.putString(r.left(),    r.bottom()-1, "╰");
    g.putString(r.right()-1, r.bottom()-1, "╯");
    // 水平线
    for (int x=r.left()+1; x<r.right()-1; x++) {
      g.putString(x, r.top(),        "─");
      g.putString(x, r.bottom()-1, "─");
    }
    // 垂直线
    for (int y=r.top()+1; y<r.bottom()-1; y++) {
      g.putString(r.left(),    y, "│");
      g.putString(r.right()-1, y, "│");
    }
    // 标题嵌入顶部边框
    if (title != null) {
      g.setForegroundColor(TextColor.ANSI.CYAN);
      g.putString(r.left()+2, r.top(),
                  " "+title+" ");
    }
  }

  /** 消息气泡 — 自己右对齐绿色，对方左对齐白色 */
  public static void drawBubble(TextGraphics g,
      Message msg, int y, int panelWidth) {
    boolean self = msg.isSelf();
    String  text = msg.content();
    String  line = self
      ? "[你] "+text
      : "["+msg.sender()+"] "+text;
    g.setForegroundColor(self
      ? TextColor.ANSI.GREEN
      : TextColor.ANSI.WHITE);
    int x = self
      ? panelWidth-line.length()-1
      : 1;
    g.putString(x, y, line);
  }
}
